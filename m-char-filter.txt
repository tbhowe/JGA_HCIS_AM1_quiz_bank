(input as nullable text) as nullable text =>
let
Safe = if input = null then "" else input,
Chars = Text.ToList(Safe),
Kept = List.Select(Chars, each
let c = Character.ToNumber(_) in
not (
c < 32                              // C0 controls (tab, LF, CR, null, etc.)
or c = 127                          // DEL
or (c >= 128 and c <= 159)          // C1 controls
or c = 173                          // soft hyphen (renders as nothing)
or (c >= 8203 and c <= 8207)        // zero-width space, ZWNJ, ZWJ, LRM, RLM
or (c >= 8232 and c <= 8238)        // line/para separators + directional overrides
or (c >= 8288 and c <= 8303)        // word joiner + invisible operators
or c = 65279                        // BOM / zero-width no-break space
or (c >= 65520 and c <= 65535)      // specials + noncharacters
)
),
Cleaned = Text.Combine(Kept),
Spaced = List.Accumulate(
{160, 5760, 6158, 8192, 8193, 8194, 8195, 8196, 8197, 8198,
8199, 8200, 8201, 8202, 8239, 8287, 12288},
Cleaned,
(state, code) => Text.Replace(state, Character.FromNumber(code), " ")
),
Result = Text.Trim(Spaced)
in
if Result = "" then null else Result
